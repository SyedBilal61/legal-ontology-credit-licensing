🟢 BASIC QUERIES



1️⃣ Get all agents
SELECT ?agent
WHERE {
  ?agent a <http://www.example.org/credit-licensing-ontology#Agent> .
}



2️⃣ Get all individuals in the ontology
SELECT ?ind
WHERE {
  ?ind a ?type .
}



3️⃣ Get all agents engaged in credit activity
SELECT ?agent
WHERE {
  ?agent <http://creditontology.com/section29#hasEngagedIn> ?activity .
}



🟢 COMPLIANCE QUERIES



4️⃣ Get compliant agents (main result)
SELECT ?agent
WHERE {
  ?agent a <http://creditontology.com/section29#CompliantAgent> .
}



5️⃣ Get non-compliant agents
SELECT ?agent
WHERE {
  ?agent a <http://creditontology.com/section29#NonCompliantAgent> .
}



6️⃣ Get agents with valid license
SELECT ?agent
WHERE {
  ?agent <http://www.example.org/credit-licensing-ontology#isValidLicense> true .
}



7️⃣ Get agents without license
SELECT ?agent
WHERE {
  ?agent <http://www.example.org/credit-licensing-ontology#isValidLicense> false .
}



🟢 LEGAL LOGIC QUERIES 



8️⃣ Agents committing offences
SELECT ?agent

9️⃣ Agents with penalties

  ?agent <http://www.co-ode.org/ontologies/ont.owl#getPenalty> true .

SELECT ?agent
WHERE {
  ?agent <http://www.co-ode.org/ontologies/ont.owl#hasOffence> true .
  ?agent <http://www.co-ode.org/ontologies/ont.owl#hasDefence> false .
}
